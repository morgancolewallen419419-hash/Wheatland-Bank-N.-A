    // Exact integer cents formatting -> USD ($XX,XXX,XXX.XX)
    function formatCentsToUSD(cents) {
      const safeCents = Math.round(Number(cents) || 0);
      const isNegative = safeCents < 0;
      const absCents = Math.abs(safeCents);
      const dollars = Math.floor(absCents / 100);
      const remainderCents = String(absCents % 100).padStart(2, '0');
      const formattedDollars = dollars.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ',');
      return `${isNegative ? '-' : ''}$${formattedDollars}.${remainderCents}`;
    }

    // Parse dollar string safely to integer cents (no floating-point drift)
    function parseDollarsToCents(input) {
      if (typeof input === 'number') return Math.round(input * 100);
      const cleaned = String(input).replace(/[^0-9.]/g, '');
      if (!cleaned) return 0;
      const parts = cleaned.split('.');
      if (parts.length > 2) return NaN;
      const whole = parseInt(parts[0] || '0', 10);
      const fracStr = (parts[1] || '').padEnd(2, '0').slice(0, 2);
      if (parts[1] && parts[1].length > 2) return NaN;
      const frac = parseInt(fracStr || '0', 10);
      return whole * 100 + frac;
    }

    // Format YYYY-MM-DD to MM/DD/YYYY (US Format)
    function formatDateUS(isoDate) {
      if (!isoDate) return '';
      const [y, m, d] = isoDate.split('T')[0].split('-');
      if (!y || !m || !d) return isoDate;
      return `${m}/${d}/${y}`;
    }

    // Format phone to +1 (XXX) XXX-XXXX
    function formatPhoneUS(raw) {
      const digits = String(raw).replace(/\D/g, '');
      const ten = digits.length === 11 && digits.startsWith('1') ? digits.slice(1) : digits.slice(0, 10);
      if (ten.length < 10) return raw;
      return `+1 (${ten.slice(0, 3)}) ${ten.slice(3, 6)}-${ten.slice(6, 10)}`;
    }

    // Mask phone number: +1 (312) •••-8492
    function maskPhoneUS(phone) {
      const formatted = formatPhoneUS(phone);
      const match = formatted.match(/^\+1 \((\d{3})\) (\d{3})-(\d{4})$/);
      if (!match) return '+1 (•••) •••-••••';
      return `+1 (${match[1]}) •••-${match[3]}`;
    }

        ch = str.charCodeAt(i);
        h1 = Math.imul(h1 ^ ch, 2654435761);
        h2 = Math.imul(h2 ^ ch, 1597334677);
      }
      h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);
      h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909);
      return `argon2id$v=19$m=65536,t=3,p=4$${(4294967296 * (2097151 & h2) + (h1 >>> 0)).toString(16)}`;
    }

    function StatusBadge({ status }) {
      const s = String(status || '').toLowerCase();
      let cls = 'bg-slate-100 text-slate-700';
      if (s === 'completed' || s === 'active' || s === 'verified' || s === 'resolved') {
        cls = 'bg-emerald-50 text-emerald-700 border border-emerald-200';
      } else if (s === 'pending' || s === 'open' || s === 'in review') {
        cls = 'bg-amber-50 text-amber-700 border border-amber-200';
      } else if (s === 'failed' || s === 'rejected' || s === 'locked') {
        cls = 'bg-rose-50 text-rose-700 border border-rose-200';
      }
      return (
        <span className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium ${cls}`}>
          {status}
        </span>
      );
    }

        dob: '',
        email: '',
        phone: '',
        street: '',
        apt: '',

        password: '',
        confirmPassword: '',
        securityQuestion: SECURITY_QUESTIONS[0],
        securityAnswer: '',
        acceptTerms: false,
        acceptPrivacy: false
      });

      const [showPwd, setShowPwd] = useState(false);
      const [showConfirmPwd, setShowConfirmPwd] = useState(false);
      const [reviewMode, setReviewMode] = useState(false);
      const [errors, setErrors] = useState({});
      const [submitting, setSubmitting] = useState(false);

      // Password validation rules
      const pwdChecks = useMemo(() => {
        const p = form.password;
        return {
          length: p.length >= 12,
          upper: /[A-Z]/.test(p),
          lower: /[a-z]/.test(p),
          number: /[0-9]/.test(p),
          special: /[^A-Za-z0-9]/.test(p)
        };
      }, [form.password]);

      const strengthScore = Object.values(pwdChecks).filter(Boolean).length;

      const updateField = (key, value) => {
        setForm(prev => ({ ...prev, [key]: value }));
        if (errors[key]) {
          setErrors(prev => {
            const copy = { ...prev };
            delete copy[key];
            return copy;
          });
        }
      };

      const validateAll = () => {
        const errs = {};
        if (!form.firstName.trim()) errs.firstName = 'First name is required.';
        if (!form.middleName.trim()) errs.middleName = 'Middle name is required.';
        if (!form.lastName.trim()) errs.lastName = 'Last name is required.';
        if (!form.dob) errs.dob = 'Date of birth is required.';
        if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) {
          errs.email = 'Please enter a valid email address.';
        } else if (existingUsers.some(u => u.email.toLowerCase() === form.email.trim().toLowerCase())) {
          errs.email = 'An account with this email address already exists.';
        }

        const digits = form.phone.replace(/\D/g, '');
        if (digits.length < 10) errs.phone = 'Enter a valid 10-digit US mobile number.';

        if (!form.street.trim()) errs.street = 'Street address is required.';
        if (!form.city.trim()) errs.city = 'City is required.';
        if (!form.state) errs.state = 'State is required.';
        if (!/^\d{5}(-\d{4})?$/.test(form.zip.trim())) errs.zip = 'Enter a valid 5-digit US ZIP code.';

        if (!form.username.trim() || form.username.trim().length < 4) {
          errs.username = 'Username must be at least 4 characters.';
        } else if (existingUsers.some(u => u.username.toLowerCase() === form.username.trim().toLowerCase())) {
          errs.username = 'This username is already taken.';
        }

        if (strengthScore < 5) {
          errs.password = 'Password must meet all 5 security requirements below.';
        }
        if (form.password !== form.confirmPassword) {
          errs.confirmPassword = 'Passwords do not match.';
        }
        if (!form.securityAnswer.trim() || form.securityAnswer.trim().length < 2) {
          errs.securityAnswer = 'Security answer is required.';
        }
        if (!form.acceptTerms) errs.acceptTerms = 'You must accept the Wheatland Bank Account Agreement.';
        if (!form.acceptPrivacy) errs.acceptPrivacy = 'You must acknowledge the U.S. Consumer Privacy Notice.';

        setErrors(errs);

    function PhoneVerificationPage({ navigate, pendingAuth, onVerifySuccess, onResendOtp }) {
      const [digits, setDigits] = useState(['', '', '', '', '', '']);
      const [error, setError] = useState('');
      const [secondsLeft, setSecondsLeft] = useState(300);
      const [resendCooldown, setResendCooldown] = useState(30);
      const inputRefs = useRef([]);

      useEffect(() => {
        if (!pendingAuth) {
          navigate('/login');
          return;
        }
        const updateTimers = () => {
          const remaining = Math.max(0, Math.floor((pendingAuth.expiresAt - Date.now()) / 1000));
          setSecondsLeft(remaining);
          const cd = Math.max(0, Math.floor((pendingAuth.resendAvailableAt - Date.now()) / 1000));
          setResendCooldown(cd);
        };
        updateTimers();
        const timer = setInterval(updateTimers, 1000);
        return () => clearInterval(timer);
      }, [pendingAuth, navigate]);

      if (!pendingAuth) return null;

      const handleDigitChange = (idx, val) => {
        const clean = val.replace(/\D/g, '');
        if (!clean) {
          const next = [...digits];
          next[idx] = '';
          setDigits(next);
          return;
        }
        // Handle multi-digit paste or typing
        if (clean.length > 1) {
          const chars = clean.slice(0, 6).split('');
          const next = ['', '', '', '', '', ''];
          chars.forEach((c, i) => { next[i] = c; });
          setDigits(next);
          const focusIdx = Math.min(5, chars.length);
          inputRefs.current[focusIdx]?.focus();
          return;
        }
        const next = [...digits];
        next[idx] = clean;
        setDigits(next);
        setError('');
        if (idx < 5) {
          inputRefs.current[idx + 1]?.focus();
        }
      };

      const handleKeyDown = (idx, e) => {
        if (e.key === 'Backspace' && !digits[idx] && idx > 0) {
          inputRefs.current[idx - 1]?.focus();
        }
      };

      const handlePaste = (e) => {
        e.preventDefault();
        const pasted = (e.clipboardData.getData('text') || '').replace(/\D/g, '').slice(0, 6);
        if (!pasted) return;
        const next = ['', '', '', '', '', ''];
        pasted.split('').forEach((c, i) => { next[i] = c; });
        setDigits(next);
        setError('');
        inputRefs.current[Math.min(5, pasted.length - 1)]?.focus();
      };

      const handleSubmit = (e) => {
        e.preventDefault();
        const code = digits.join('');
        if (code.length !== 6) {
          setError('Please enter all 6 digits of your verification code.');
          return;
        }
        if (Date.now() > pendingAuth.expiresAt) {
          setError('This verification code has expired (5-minute limit). Please request a new code.');
          return;
        }
        if (pendingAuth.attempts >= 5) {
          setError('Maximum verification attempts exceeded. Please request a new code.');
          return;
        }
        const result = onVerifySuccess(code);
        if (!result.ok) {
          setError(result.error);
          setDigits(['', '', '', '', '', '']);
          inputRefs.current[0]?.focus();
        }
      };

      const mins = String(Math.floor(secondsLeft / 60)).padStart(2, '0');
      const secs = String(secondsLeft % 60).padStart(2, '0');

      return (
        <div className="min-h-screen bg-[#F4F6F9] flex flex-col justify-center py-10 px-4">
          <div className="max-w-md w-full mx-auto space-y-6">
            <div className="text-center">
              <WheatlandLogo size="lg" />
              <h1 className="text-2xl font-bold text-[#0F294A] mt-4">SMS Phone Verification</h1>
              <p className="text-sm text-slate-600 mt-1">
                We sent a 6-digit security code to{' '}
                <strong className="text-slate-900">{maskPhoneUS(pendingAuth.user.phone)}</strong>
              </p>
            </div>

            {/* Controlled SMS Test Environment Banner so user can inspect & test OTP seamlessly */}
            <div className="bg-amber-50 border border-amber-300 rounded-2xl p-4 shadow-sm">
              <div className="flex items-center justify-between gap-2">
                <span className="text-[11px] font-bold uppercase tracking-wider text-amber-900">
                  Controlled SMS Test Gateway
                </span>
                <button
                  type="button"
                  onClick={() => {
                    const chars = pendingAuth.devCode.split('');
                    setDigits(chars);
                    setError('');
                  }}
                  className="text-xs font-semibold bg-amber-900 text-white px-2.5 py-1 rounded-lg hover:bg-amber-800"
                >
                  Auto-Fill Code
                </button>
              </div>
              <p className="text-xs text-amber-900 mt-1.5">
                Simulated SMS Delivery to {formatPhoneUS(pendingAuth.user.phone)}: Your Wheatland Bank verification code is{' '}
                <code className="font-mono font-bold text-sm bg-white px-2 py-0.5 rounded border border-amber-300">
                  {pendingAuth.devCode}
                </code>
              </p>
            </div>

            <div className="bg-white rounded-2xl border border-slate-200 shadow-sm p-6 sm:p-8 space-y-5">
              {error && (
                <div className="p-3.5 rounded-xl bg-rose-50 border border-rose-200 text-xs font-medium text-rose-800">
                  {error}
                </div>
              )}

              <form onSubmit={handleSubmit} className="space-y-5">
                <div>
                  <label className="block text-xs font-semibold text-slate-700 text-center mb-3">

    function ForgotAndResetPasswordPage({ mode, navigate, users, onResetPassword }) {
      const [emailOrUser, setEmailOrUser] = useState('');
      const [foundUser, setFoundUser] = useState(null);
      const [answer, setAnswer] = useState('');
      const [newPassword, setNewPassword] = useState('');
      const [confirmPassword, setConfirmPassword] = useState('');
      const [error, setError] = useState('');
      const [step, setStep] = useState(mode === 'reset' ? 2 : 1);

      const handleLookup = (e) => {
        e.preventDefault();
        setError('');
        const q = emailOrUser.trim().toLowerCase();
        const match = users.find(u => u.email.toLowerCase() === q || u.username.toLowerCase() === q);
        if (!match) {
          setError('No customer profile found matching that email address or username.');
          return;
        }
        setFoundUser(match);
        setStep(2);
      };

      const handleReset = (e) => {
        e.preventDefault();
        setError('');
        const targetUser = foundUser || users[0];
        if (!targetUser) {
          setError('Please locate your account first.');
          setStep(1);
          return;
        }
        if (secureHashSim(answer.trim().toLowerCase()) !== targetUser.securityAnswerHash) {
          setError('Security question answer does not match our records.');
          return;
        }
        const validPwd =
          newPassword.length >= 12 &&
          /[A-Z]/.test(newPassword) &&
          /[a-z]/.test(newPassword) &&
          /[0-9]/.test(newPassword) &&
          /[^A-Za-z0-9]/.test(newPassword);

        if (!validPwd) {
          setError('New password must be at least 12 characters with uppercase, lowercase, number, and special character.');
          return;
        }
        if (newPassword !== confirmPassword) {
          setError('Password confirmation does not match.');
          return;
        }
        onResetPassword(targetUser.id, newPassword);
      };

      return (
        <div className="min-h-screen bg-[#F4F6F9] flex flex-col justify-center py-10 px-4">
          <div className="max-w-md w-full mx-auto space-y-6">
            <div className="text-center">
              <WheatlandLogo size="lg" />
              <h1 className="text-2xl font-bold text-[#0F294A] mt-4">
                {step === 1 ? 'Account Recovery' : 'Reset Online Banking Password'}
              </h1>
            </div>

            <div className="bg-white rounded-2xl border border-slate-200 shadow-sm p-6 sm:p-8 space-y-5">
              {error && (
                <div className="p-3.5 rounded-xl bg-rose-50 border border-rose-200 text-xs font-medium text-rose-800">
                  {error}
                </div>
              )}

              {step === 1 ? (
                <form onSubmit={handleLookup} className="space-y-4">
                  <p className="text-xs text-slate-600 leading-relaxed">
                    Enter your registered email address or username to verify your identity via your security question.
                  </p>
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 mb-1">Email or Username</label>
                    <input
                      type="text"
                      value={emailOrUser}
                      onChange={e => setEmailOrUser(e.target.value)}
                      className="w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm"
                      placeholder="jhenderson"
                    />
                  </div>
                  <button
                    type="submit"
                    className="w-full py-3 bg-[#0F294A] hover:bg-[#13315C] text-white font-semibold rounded-xl text-sm"
                  >
                    Continue to Security Verification
                  </button>
                </form>
              ) : (
                <form onSubmit={handleReset} className="space-y-4">
                  <div className="p-3.5 rounded-xl bg-slate-50 border border-slate-200 text-xs">
                    <span className="text-slate-500 block">Security Question for {(foundUser || users[0])?.username}:</span>
                    <strong className="text-slate-900 mt-1 block">{(foundUser || users[0])?.securityQuestion}</strong>
                    <span className="text-[11px] text-slate-400 mt-1 block">(Demo hint: Lincoln Academy)</span>
                  </div>
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 mb-1">Security Answer *</label>
                    <input
                      type="text"
                      value={answer}
                      onChange={e => setAnswer(e.target.value)}
                      className="w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm"
                      placeholder="Enter your answer"
                    />
                  </div>
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 mb-1">New Password (12+ chars) *</label>
                    <input
                      type="password"
                      value={newPassword}
                      onChange={e => setNewPassword(e.target.value)}
                      className="w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm"
                      placeholder="New strong password"
                    />
                  </div>
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 mb-1">Confirm New Password *</label>
                    <input
                      type="password"
                      value={confirmPassword}
                      onChange={e => setConfirmPassword(e.target.value)}
                      className="w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm"
                      placeholder="Confirm password"
                    />
                  </div>
                  <button
                    type="submit"
                    className="w-full py-3 bg-[#0F294A] hover:bg-[#13315C] text-white font-semibold rounded-xl text-sm"
                  >
                    Update Password &amp; Return to Login
                  </button>
                </form>
              )}

              <div className="text-center pt-2 border-t border-slate-100">
                <button onClick={() => navigate('/login')} className="text-xs font-semibold text-slate-600 hover:text-[#0F294A]">
                  &larr; Back to Sign In
                </button>
              </div>
            </div>
          </div>
        </div>
      );
    }

    function TransactionsPage({ accounts, transactions }) {
      const [search, setSearch] = useState('');
      const [accountFilter, setAccountFilter] = useState('ALL');
      const [categoryFilter, setCategoryFilter] = useState('ALL');
      const [directionFilter, setDirectionFilter] = useState('ALL');
      const [sortOrder, setSortOrder] = useState('NEWEST');
      const [page, setPage] = useState(1);
      const [selectedTx, setSelectedTx] = useState(null);
      const pageSize = 8;

      const getAccountName = (id) => accounts.find(a => a.id === id)?.name || id;

      const filtered = useMemo(() => {
        // Strictly enforce November 2, 2026 cutoff
        let list = transactions.filter(t => t.date <= CUTOFF_DATE_STR);

        if (accountFilter !== 'ALL') {
          list = list.filter(t => t.accountId === accountFilter);
        }
        if (categoryFilter !== 'ALL') {
          list = list.filter(t => t.category === categoryFilter);
        }
        if (directionFilter !== 'ALL') {
          list = list.filter(t => t.direction === directionFilter);
        }
        if (search.trim()) {
          const q = search.trim().toLowerCase();
          list = list.filter(t =>
            t.description.toLowerCase().includes(q) ||
            t.id.toLowerCase().includes(q) ||
            t.category.toLowerCase().includes(q)
          );
        }

        // Chronological array is stored oldest-to-newest; reverse for newest-first default
        const ordered = sortOrder === 'NEWEST' ? [...list].reverse() : [...list];
        return ordered;
      }, [transactions, accountFilter, categoryFilter, directionFilter, search, sortOrder]);

      const totalPages = Math.max(1, Math.ceil(filtered.length / pageSize));
      const paginated = filtered.slice((page - 1) * pageSize, page * pageSize);

      useEffect(() => {
        setPage(1);
      }, [search, accountFilter, categoryFilter, directionFilter, sortOrder]);

      return (
        <div className="space-y-6">
          <div className="flex flex-wrap items-center justify-between gap-2">
            <div>
              <h1 className="text-2xl font-bold text-[#0F294A]">Transaction History &amp; Ledger</h1>
              <p className="text-xs text-slate-500">
                Showing verified account activity through <strong>November 2, 2026</strong> (Newest first)
              </p>
            </div>
            <span className="px-3 py-1.5 rounded-xl bg-blue-50 border border-blue-200 text-xs font-semibold text-[#0F294A]">
              Cutoff Date Enforced: 11/02/2026
            </span>
          </div>

          {/* Search & Filters */}
          <div className="bg-white rounded-2xl border border-slate-200 p-4 sm:p-5 shadow-sm grid sm:grid-cols-2 lg:grid-cols-5 gap-3">
            <div className="lg:col-span-2">
              <label className="block text-[11px] font-bold uppercase text-slate-500 mb-1">Search Ledger</label>
              <input
                type="text"
                value={search}
                onChange={e => setSearch(e.target.value)}
                placeholder="Search description, TXN ID, or merchant..."
                className="w-full rounded-xl border border-slate-300 px-3.5 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
              />
            </div>
            <div>
              <label className="block text-[11px] font-bold uppercase text-slate-500 mb-1">Account</label>
              <select

    function DepositPage({ accounts, onExecuteDeposit, navigate }) {
      const [accountId, setAccountId] = useState(accounts[0]?.id || '');
      const [amountStr, setAmountStr] = useState('');
      const [category, setCategory] = useState('Incoming deposits');
      const [description, setDescription] = useState('');
      const [date, setDate] = useState(CUTOFF_DATE_STR);
      const [step, setStep] = useState('FORM'); // FORM | REVIEW | CONFIRMED
      const [error, setError] = useState('');
      const [receipt, setReceipt] = useState(null);

      const selectedAccount = accounts.find(a => a.id === accountId) || accounts[0];
      const amountCents = parseDollarsToCents(amountStr);

      const handleReview = (e) => {
        e.preventDefault();
        setError('');
        if (isNaN(amountCents) || amountCents <= 0) {
          setError('Please enter a valid positive USD amount (e.g., 2500.00).');
          return;
        }
        if (amountCents > 5000000000) {
          setError('Single simulated deposit cannot exceed $50,000,000.00.');
          return;
        }
        if (!description.trim()) {
          setError('Please provide a deposit description or memo.');
          return;
        }
        if (!date || date > CUTOFF_DATE_STR) {
          setError('In this educational environment, ledger deposit dates must be on or before November 2, 2026.');
          return;
        }
        setStep('REVIEW');
      };

      const handleConfirm = () => {
        const tx = onExecuteDeposit({
          accountId: selectedAccount.id,
          amountCents,
          category,
          description: description.trim(),
          date
        });
        setReceipt(tx);
        setStep('CONFIRMED');
      };

      return (
        <div className="max-w-2xl mx-auto space-y-6">
          <div>
            <span className="text-xs font-bold uppercase tracking-wider text-blue-600">Simulated Educational Deposit</span>

    function TransferPage({ accounts, onExecuteTransfer, navigate }) {
      const [sourceId, setSourceId] = useState(accounts[0]?.id || '');
      const [destId, setDestId] = useState(accounts[1]?.id || '');
      const [amountStr, setAmountStr] = useState('');
      const [description, setDescription] = useState('');
      const [step, setStep] = useState('FORM');
      const [error, setError] = useState('');
      const [receipt, setReceipt] = useState(null);

      const sourceAcc = accounts.find(a => a.id === sourceId) || accounts[0];
      const destAcc = accounts.find(a => a.id === destId) || accounts[1];
      const amountCents = parseDollarsToCents(amountStr);

      const handleSourceChange = (newSrc) => {
        setSourceId(newSrc);
        if (newSrc === destId) {
          const other = accounts.find(a => a.id !== newSrc);
          if (other) setDestId(other.id);
        }
      };

      const handleReview = (e) => {
        e.preventDefault();
        setError('');
        if (sourceId === destId) {
          setError('Source and destination accounts must be different.');
          return;
        }
        if (isNaN(amountCents) || amountCents <= 0) {
          setError('Please enter a valid positive transfer amount.');
          return;
        }
        if (amountCents > sourceAcc.balanceCents) {
          setError(`Insufficient funds in ${sourceAcc.name}. Available balance: ${formatCentsToUSD(sourceAcc.balanceCents)}.`);
          return;
        }
        setStep('REVIEW');
      };

      const handleConfirm = () => {
        const res = onExecuteTransfer({
          sourceAccountId: sourceAcc.id,
          destAccountId: destAcc.id,
          amountCents,
          description: description.trim() || `Internal Transfer: ${sourceAcc.name} to ${destAcc.name}`
        });
        if (!res.ok) {
          setError(res.error);
          setStep('FORM');
          return;
        }
        setReceipt(res);
        setStep('CONFIRMED');
      };

      return (
        <div className="max-w-2xl mx-auto space-y-6">
          <div>
            <span className="text-xs font-bold uppercase tracking-wider text-blue-600">Atomic Internal Ledger Transfer</span>
            <h1 className="text-2xl font-bold text-[#0F294A]">Transfer Between Accounts</h1>
            <p className="text-xs text-slate-500 mt-0.5">
              Debits the source account and credits the destination account by the exact same integer-cent amount.
            </p>
          </div>

          <div className="bg-white rounded-2xl border border-slate-200 p-6 sm:p-8 shadow-sm">
            {step === 'FORM' && (
              <form onSubmit={handleReview} className="space-y-4">
                {error && (
                  <div className="p-3.5 rounded-xl bg-rose-50 border border-rose-200 text-xs font-medium text-rose-800">
                    {error}
                  </div>
                )}

                <div className="grid sm:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 mb-1">1. From Source Account *</label>
                    <select
                      value={sourceId}
                      onChange={e => handleSourceChange(e.target.value)}
                      className="w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm bg-white"
                    >
                      {accounts.map(a => (
                        <option key={a.id} value={a.id}>
                          {a.name} ({formatCentsToUSD(a.balanceCents)})
                        </option>
                      ))}
                    </select>
                  </div>

                  <div>
                    <label className="block text-xs font-semibold text-slate-700 mb-1">2. To Destination Account *</label>
                    <select
                      value={destId}
                      onChange={e => setDestId(e.target.value)}
                      className="w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm bg-white"
                    >
                      {accounts.map(a => (
                        <option key={a.id} value={a.id} disabled={a.id === sourceId}>
                          {a.name} ({formatCentsToUSD(a.balanceCents)})
                        </option>
                      ))}
                    </select>
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-700 mb-1">3. Transfer Amount (USD) *</label>
                  <input
                    type="text"
                    inputMode="decimal"
                    value={amountStr}
                    onChange={e => setAmountStr(e.target.value)}
                    placeholder="10000.00"
                    className="w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm font-mono"
                  />
                  <p className="text-[11px] text-slate-500 mt-1">
                    Available in {sourceAcc.name}: <strong>{formatCentsToUSD(sourceAcc.balanceCents)}</strong>
                  </p>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-700 mb-1">Transfer Memo (Optional)</label>
                  <input
                    type="text"
                    value={description}
                    onChange={e => setDescription(e.target.value)}
                    placeholder="e.g., Monthly Reserve Rebalancing"
                    className="w-full rounded-xl border border-slate-300 px-3.5 py-2.5 text-sm"
                  />
                </div>

                <div className="pt-3 flex justify-end">
                  <button
                    type="submit"
                    className="px-6 py-3 bg-[#0F294A] hover:bg-[#13315C] text-white text-sm font-semibold rounded-xl shadow-sm"
                  >
                    Review Transfer &rarr;
                  </button>
                </div>
              </form>
            )}

            {step === 'REVIEW' && (
              <div className="space-y-5">
                <h2 className="text-lg font-bold text-[#0F294A]">Review Internal Transfer</h2>
                <div className="bg-slate-50 rounded-xl p-5 border border-slate-200 space-y-3 text-sm">
                  <div className="flex justify-between">
                    <span className="text-slate-500">From Account</span>
                    <strong>{sourceAcc.name}</strong>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-500">To Account</span>
                    <strong>{destAcc.name}</strong>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-500">Transfer Amount</span>
                    <strong className="font-mono text-base text-[#0F294A]">{formatCentsToUSD(amountCents)}</strong>
                  </div>
                  <div className="border-t border-slate-200 pt-2 flex justify-between text-xs">
                    <span className="text-slate-500">New {sourceAcc.name} Balance</span>
                    <span className="font-mono font-semibold">{formatCentsToUSD(sourceAcc.balanceCents - amountCents)}</span>
                  </div>
                  <div className="flex justify-between text-xs">
                    <span className="text-slate-500">New {destAcc.name} Balance</span>
                    <span className="font-mono font-semibold">{formatCentsToUSD(destAcc.balanceCents + amountCents)}</span>
                  </div>
                </div>

                <div className="flex items-center justify-between pt-2">
                  <button
                    onClick={() => setStep('FORM')}
                    className="px-4 py-2.5 border border-slate-300 rounded-xl text-xs font-semibold text-slate-700"
                  >
                    &larr; Modify
                  </button>
                  <button
                    onClick={handleConfirm}
                    className="px-6 py-3 bg-[#0F294A] hover:bg-[#13315C] text-white text-sm font-semibold rounded-xl shadow"
                  >
                    Confirm &amp; Execute Transfer
                  </button>
                </div>
              </div>
            )}

            {step === 'CONFIRMED' && receipt && (
              <div className="space-y-5 text-center">
                <div className="w-12 h-12 rounded-full bg-emerald-100 text-emerald-700 flex items-center justify-center mx-auto text-xl font-bold">
                  ✓
                </div>
                <div>
                  <h2 className="text-xl font-bold text-[#0F294A]">Atomic Transfer Completed</h2>
                  <p className="text-xs text-slate-500 mt-1">Reference: <span className="font-mono font-semibold">{receipt.transferId}</span></p>
                </div>
                <div className="bg-slate-50 rounded-xl p-4 border border-slate-200 text-sm space-y-2 text-left">
                  <div className="flex justify-between">
                    <span className="text-slate-500">Amount Transferred</span>
                    <strong className="font-mono text-[#0F294A]">{formatCentsToUSD(receipt.amountCents)}</strong>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-500">{sourceAcc.name} New Balance</span>
                    <strong className="font-mono">{formatCentsToUSD(receipt.debitTx.balanceAfterCents)}</strong>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-500">{destAcc.name} New Balance</span>
                    <strong className="font-mono">{formatCentsToUSD(receipt.creditTx.balanceAfterCents)}</strong>
                  </div>
                </div>
                <div className="flex flex-wrap justify-center gap-3 pt-2">
                  <button
                    onClick={() => { setAmountStr(''); setDescription(''); setStep('FORM'); }}
                    className="px-4 py-2.5 border border-slate-300 rounded-xl text-xs font-semibold text-slate-700"
                  >
                    Make Another Transfer
                  </button>
                  <button
                    onClick={() => navigate('/dashboard')}
                    className="px-5 py-2.5 bg-[#0F294A] text-white rounded-xl text-xs font-semibold"
                  >
                    Return to Dashboard
                  </button>
                </div>
              </div>
            )}
          </div>
        </div>
      );
    }

    function StatementsPage({ currentUser, accounts, transactions }) {
      const [accountId, setAccountId] = useState(accounts[0]?.id || '');

      const account = accounts.find(a => a.id === accountId) || accounts[0];

      const statementData = useMemo(() => {
        const allAccTx = transactions.filter(t => t.accountId === account.id && t.date <= CUTOFF_DATE_STR);
        const periodTx = allAccTx.filter(t => t.date.startsWith(selectedMonth));

        // Opening balance before the selected month
        let openingCents = account.openingBalanceCents;
        allAccTx.forEach(t => {
          if (t.date < `${selectedMonth}-01`) {
            openingCents += t.direction === 'CREDIT' ? t.amountCents : -t.amountCents;
          }
        });

        let totalCredits = 0;
        let totalDebits = 0;
        periodTx.forEach(t => {
          if (t.direction === 'CREDIT') totalCredits += t.amountCents;
          else totalDebits += t.amountCents;
        });

        const closingCents = openingCents + totalCredits - totalDebits;

          openingCents,
          totalCredits,
          totalDebits,
          closingCents,
          periodTx
        };

    function SecurityAndSettingsPage({

      onSignOutAllDevices
    }) {
      const [currentPwd, setCurrentPwd] = useState('');
      const [newPwd, setNewPwd] = useState('');
      const [confirmPwd, setConfirmPwd] = useState('');
      const [pwdError, setPwdError] = useState('');

      const handlePwdSubmit = (e) => {
        e.preventDefault();
        setPwdError('');
        if (secureHashSim(currentPwd) !== currentUser.passwordHash) {
          setPwdError('Current password is incorrect.');
          return;
        }
        const valid =
          newPwd.length >= 12 &&
          /[A-Z]/.test(newPwd) &&
          /[a-z]/.test(newPwd) &&
          /[0-9]/.test(newPwd) &&
          /[^A-Za-z0-9]/.test(newPwd);
        if (!valid) {
          setPwdError('New password must be at least 12 characters with uppercase, lowercase, number, and symbol.');
          return;
        }
        if (newPwd !== confirmPwd) {
          setPwdError('New password confirmation does not match.');
          return;
        }
        onChangePassword(newPwd);
        setCurrentPwd('');
        setNewPwd('');
        setConfirmPwd('');

        <div className="max-w-4xl mx-auto space-y-6">

                    <p>Address: {selectedCust.street}, {selectedCust.city}, {selectedCust.state} {selectedCust.zip}</p>
                    <p>Member Since: {formatDateUS(selectedCust.membershipDate)} &bull; DOB: {formatDateUS(selectedCust.dob)}</p>

                  <h2 className="text-lg font-bold text-[#0F294A]">Master Financial Ledger Inspector</h2>

                <div className="overflow-x-auto">
                  <table className="w-full text-left border-collapse text-xs">
                    <thead>
                      <tr className="bg-slate-50 border-b border-slate-200 font-bold text-slate-600">
                        <th className="py-3 px-4">TXN ID</th>
                        <th className="py-3 px-4">Date &amp; Time</th>
                        <th className="py-3 px-4">Description</th>
                        <th className="py-3 px-4">Direction</th>
                        <th className="py-3 px-4 text-right">Amount</th>
                        <th className="py-3 px-4 text-right">Post-Balance</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-100">
                      {filteredTransactions.map(tx => (
                        <tr key={tx.id} className="hover:bg-slate-50">
                          <td className="py-3 px-4 font-mono font-semibold">{tx.id}</td>
                          <td className="py-3 px-4">{formatDateUS(tx.date)} {tx.time}</td>
                          <td className="py-3 px-4 font-medium">{tx.description}</td>
                          <td className="py-3 px-4">{tx.direction}</td>
                          <td className="py-3 px-4 text-right font-mono font-bold">
                            {tx.direction === 'CREDIT' ? '+' : '-'}{formatCentsToUSD(tx.amountCents)}
                          </td>
                          <td className="py-3 px-4 text-right font-mono">{formatCentsToUSD(tx.balanceAfterCents)}</td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              </div>
            )}

            {route === '/admin/audit-logs' && (
              <div className="bg-white rounded-2xl border border-slate-200 p-6 shadow-sm space-y-4">
                <div>
                  <h2 className="text-lg font-bold text-[#0F294A]">Immutable System Audit Logs</h2>
                  <p className="text-xs text-slate-500">Every authentication event, ledger operation, profile edit, and administrative action is recorded.</p>
                </div>
                <div className="overflow-x-auto">
                  <table className="w-full text-left border-collapse text-xs">
                    <thead>
                      <tr className="bg-slate-50 border-b border-slate-200 font-bold text-slate-600">
                        <th className="py-3 px-4">Audit ID</th>
                        <th className="py-3 px-4">Timestamp</th>
                        <th className="py-3 px-4">Actor</th>
                        <th className="py-3 px-4">Category</th>
                        <th className="py-3 px-4">Event &amp; Details</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-100">
                      {state.auditLogs.map(log => (
                        <tr key={log.id} className="hover:bg-slate-50">
                          <td className="py-3 px-4 font-mono font-semibold text-slate-700">{log.id}</td>
                          <td className="py-3 px-4 whitespace-nowrap">{log.timestamp}</td>
                          <td className="py-3 px-4 font-semibold">{log.actor}</td>
                          <td className="py-3 px-4"><StatusBadge status={log.category} /></td>
                          <td className="py-3 px-4">
                            <p className="font-bold text-slate-900">{log.action}</p>
                            <p className="text-slate-500">{log.details}</p>
                          </td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              </div>
            )}
          </main>
        </div>
      );
    }

    /* ============================================================================
       8. MAIN ROOT APPLICATION CONTROLLER & ROUTER
       ============================================================================ */
    function App() {
      const [bankState, setBankState] = useState(() => {
        try {
          const saved = localStorage.getItem(STORAGE_KEY);
          if (saved) return JSON.parse(saved);
        } catch (e) {
          console.warn('Storage unavailable, using fresh seed state');
        }
        return buildInitialSeedState();
      });

      const [route, setRoute] = useState('/');
      const [authenticatedUserId, setAuthenticatedUserId] = useState(null);
      const [pendingAuth, setPendingAuth] = useState(null);
      const [isAdminAuthenticated, setIsAdminAuthenticated] = useState(false);
      const [toast, setToast] = useState(null);

      useEffect(() => {
        try {
          localStorage.setItem(STORAGE_KEY, JSON.stringify(bankState));
        } catch (e) {
          // Ignore storage quota errors
        }
      }, [bankState]);

      const showToast = useCallback((type, title, message = '') => {
        setToast({ type, title, message });
        setTimeout(() => {
          setToast(prev => (prev && prev.title === title ? null : prev));
        }, 4500);
      }, []);

      const appendAuditLog = useCallback((actor, category, action, details) => {
        setBankState(prev => ({
          ...prev,
          auditLogs: [
            {
              id: generateId('AUD'),
              timestamp: `${formatDateUS(CUTOFF_DATE_STR)} ${getCurrentChicagoTime12h()}`,
              actor,
              category,
              action,
              details
            },
            ...prev.auditLogs
          ]
        }));
      }, []);

      const pushNotification = useCallback((userId, type, title, message) => {
        setBankState(prev => ({
          ...prev,
          notifications: [
            {
              id: generateId('NTF'),
              userId,
              type,
              title,
              message,
              date: formatDateUS(CUTOFF_DATE_STR),
              time: getCurrentChicagoTime12h(),
              read: false
            },
            ...prev.notifications
          ]
        }));
      }, []);

      const navigate = useCallback((targetPath) => {
        setRoute(targetPath);
        window.scrollTo({ top: 0, behavior: 'smooth' });
      }, []);

      // Initiate 6-digit SMS OTP challenge
      const startOtpChallenge = (user) => {
        const sixDigit = String(Math.floor(100000 + Math.random() * 900000));
        setPendingAuth({
          user,
          codeHash: secureHashSim(sixDigit),
          devCode: sixDigit,
          expiresAt: Date.now() + 5 * 60 * 1000,
          resendAvailableAt: Date.now() + 30 * 1000,
          attempts: 0
        });
        appendAuditLog(user.username, 'AUTH', 'SMS Verification Code Issued', `Sent 6-digit OTP to ${maskPhoneUS(user.phone)}`);
        navigate('/verify-phone');
      };

      const handleRegisterSubmit = (formData) => {
        const newUserId = generateId('USR');
        const newChkId = generateId('ACC-CHK');
        const newSavId = generateId('ACC-SAV');
        const randSuffix1 = Math.floor(100000000000 + Math.random() * 899999999999);
        const randSuffix2 = randSuffix1 + 1;

        const newUser = {
          id: newUserId,
          username: formData.username.trim(),
          email: formData.email.trim(),
          passwordHash: secureHashSim(formData.password),
          firstName: formData.firstName.trim(),
          middleName: formData.middleName.trim(),
          lastName: formData.lastName.trim(),
          dob: formData.dob,
          phone: formData.phone,
          phoneVerified: false,
          street: formData.street.trim(),
          apt: formData.apt.trim(),
          city: formData.city.trim(),
          state: formData.state,
          zip: formData.zip.trim(),
          country: 'United States',
          securityQuestion: formData.securityQuestion,
          securityAnswerHash: secureHashSim(formData.securityAnswer.trim().toLowerCase()),
          membershipDate: CUTOFF_DATE_STR,
          status: 'Active',
          loginAlerts: true
        };

        const initialChkCents = 1978030000; // $19,780,300.00
        const initialSavCents = 228000000;  // $2,280,000.00

        const newAccounts = [
          {
            id: newChkId,
            userId: newUserId,
            type: 'Checking',
            name: 'Wheatland Checking',
            accountIdentifier: `WB-DEMO-${randSuffix1}`,
            routingPlaceholder: '000000000',
            currency: 'USD',
            openingBalanceCents: initialChkCents,
            balanceCents: initialChkCents,
            openedDate: CUTOFF_DATE_STR,
            status: 'Active',
            apy: '0.45%'
          },
          {
            id: newSavId,
            userId: newUserId,
            type: 'Savings',
            name: 'Wheatland Savings',
            accountIdentifier: `WB-DEMO-${randSuffix2}`,
            routingPlaceholder: '000000000',
            currency: 'USD',
            openingBalanceCents: initialSavCents,
            balanceCents: initialSavCents,
            openedDate: CUTOFF_DATE_STR,
            status: 'Active',
            apy: '4.35%'
          }
        ];

        const openingTx = [
          {
            id: generateId('TXN'),
            userId: newUserId,
            accountId: newChkId,
            date: CUTOFF_DATE_STR,
            time: '09:00 AM',
            description: 'Initial Wheatland Checking Account Funding',
            category: 'Incoming deposits',
            direction: 'CREDIT',
            amountCents: initialChkCents,
            balanceAfterCents: initialChkCents,
            status: 'Completed'
          },
          {
            id: generateId('TXN'),
            userId: newUserId,
            accountId: newSavId,
            date: CUTOFF_DATE_STR,
            time: '09:00 AM',
            description: 'Initial Wheatland Savings Account Funding',
            category: 'Incoming deposits',
            direction: 'CREDIT',
            amountCents: initialSavCents,
            balanceAfterCents: initialSavCents,
            status: 'Completed'
          }
        ];

        setBankState(prev => ({
          ...prev,
          users: [...prev.users, newUser],
          accounts: [...prev.accounts, ...newAccounts],
          transactions: [...prev.transactions, ...openingTx]
        }));

        pushNotification(newUserId, 'Registration', 'Welcome to Wheatland Bank', 'Your customer profile and deposit accounts have been created.');
        appendAuditLog(newUser.username, 'REGISTRATION', 'Customer Profile Created', `Provisioned Checking & Savings for ${newUser.email}`);
        showToast('success', 'Application Created', 'Please complete 6-digit SMS phone verification.');
        startOtpChallenge(newUser);
      };

      const handleVerifyOtp = (enteredCode) => {
        if (!pendingAuth) return { ok: false, error: 'Session expired.' };
        if (secureHashSim(enteredCode) !== pendingAuth.codeHash) {
          setPendingAuth(prev => ({ ...prev, attempts: prev.attempts + 1 }));
          appendAuditLog(pendingAuth.user.username, 'SECURITY', 'Failed OTP Verification Attempt', 'Incorrect 6-digit code entered');
          return { ok: false, error: 'Incorrect 6-digit verification code. Please try again.' };
        }

        const verifiedUser = { ...pendingAuth.user, phoneVerified: true };
        setBankState(prev => ({
          ...prev,
          users: prev.users.map(u => (u.id === verifiedUser.id ? verifiedUser : u))
        }));
        setAuthenticatedUserId(verifiedUser.id);
        setPendingAuth(null);
        pushNotification(verifiedUser.id, 'Login', 'Authenticated Sign-In Verified', 'Your online banking session was verified via 6-digit SMS OTP.');
        appendAuditLog(verifiedUser.username, 'AUTH', 'Customer Login Completed', '6-digit SMS verification succeeded');
        showToast('success', `Welcome, ${verifiedUser.firstName}`, 'SMS verification succeeded.');
        navigate('/dashboard');
        return { ok: true };
      };

      const handleQuickDemoLogin = () => {
        const demoUser = bankState.users[0];
        setAuthenticatedUserId(demoUser.id);
        setPendingAuth(null);
        appendAuditLog(demoUser.username, 'AUTH', 'Quick Demo Session Initiated', 'Logged into pre-configured Wheatland Bank account');
        showToast('success', `Welcome back, ${demoUser.firstName}`, 'Connected to pre-configured Wheatland Bank account.');
        navigate('/dashboard');
      };

      const handleLogout = () => {
        const user = bankState.users.find(u => u.id === authenticatedUserId);
        if (user) {
          appendAuditLog(user.username, 'AUTH', 'Customer Logged Out', 'Protected session terminated');
        }
        setAuthenticatedUserId(null);
        setPendingAuth(null);
        showToast('info', 'Signed Out', 'You have securely logged out of Wheatland Bank.');
        navigate('/');
      };

      const currentUser = useMemo(
        () => bankState.users.find(u => u.id === authenticatedUserId) || null,
        [bankState.users, authenticatedUserId]
      );

      const userAccounts = useMemo(
        () => (currentUser ? bankState.accounts.filter(a => a.userId === currentUser.id) : []),
        [bankState.accounts, currentUser]
      );

      const userTransactions = useMemo(
        () => (currentUser ? bankState.transactions.filter(t => t.userId === currentUser.id) : []),
        [bankState.transactions, currentUser]
      );

      const userNotifications = useMemo(
        () => (currentUser ? bankState.notifications.filter(n => n.userId === currentUser.id) : []),
        [bankState.notifications, currentUser]
      );

      const userTickets = useMemo(
        () => (currentUser ? bankState.supportTickets.filter(t => t.userId === currentUser.id) : []),
        [bankState.supportTickets, currentUser]
      );

      const userSessions = useMemo(
        () => (currentUser ? bankState.sessions.filter(s => s.userId === currentUser.id) : []),
        [bankState.sessions, currentUser]
      );

      // Atomic Ledger Operations (Deposit, Withdrawal, Transfer)
      const executeDeposit = ({ accountId, amountCents, category, description, date }) => {
        const targetAcc = userAccounts.find(a => a.id === accountId);
        const newBal = targetAcc.balanceCents + amountCents;
        const tx = {
          id: generateId('TXN'),
          userId: currentUser.id,
          accountId,
          date,
          time: getCurrentChicagoTime12h(),
          description,
          category,
          direction: 'CREDIT',
          amountCents,
          balanceAfterCents: newBal,
          status: 'Completed'
        };

        setBankState(prev => ({
          ...prev,
          accounts: prev.accounts.map(a => (a.id === accountId ? { ...a, balanceCents: newBal } : a)),
          transactions: [...prev.transactions, tx]
        }));

        pushNotification(
          currentUser.id,
          'Deposit',
          'Deposit Credited',
          `${formatCentsToUSD(amountCents)} deposited into ${targetAcc.name}.`
        );
        appendAuditLog(currentUser.username, 'DEPOSIT', `Simulated Deposit (+${formatCentsToUSD(amountCents)})`, `${tx.id} on ${targetAcc.accountIdentifier}`);
        showToast('success', 'Deposit Confirmed', `${formatCentsToUSD(amountCents)} added to ${targetAcc.name}.`);
        return tx;
      };

      const executeWithdrawal = ({ accountId, amountCents, description }) => {
        const targetAcc = userAccounts.find(a => a.id === accountId);
        if (!targetAcc || amountCents > targetAcc.balanceCents) {
          return { ok: false, error: 'Insufficient available balance.' };
        }
        const newBal = targetAcc.balanceCents - amountCents;
        const tx = {
          id: generateId('TXN'),
          userId: currentUser.id,
          accountId,
          date: CUTOFF_DATE_STR,
          time: getCurrentChicagoTime12h(),
          description,
          category: 'ATM withdrawals',
          direction: 'DEBIT',
          amountCents,
          balanceAfterCents: newBal,
          status: 'Completed'
        };

        setBankState(prev => ({
          ...prev,
          accounts: prev.accounts.map(a => (a.id === accountId ? { ...a, balanceCents: newBal } : a)),
          transactions: [...prev.transactions, tx]
        }));

        pushNotification(
          currentUser.id,
          'Withdrawal',
          'Withdrawal Processed',
          `${formatCentsToUSD(amountCents)} withdrawn from ${targetAcc.name}.`
        );
        appendAuditLog(currentUser.username, 'WITHDRAWAL', `Withdrawal (-${formatCentsToUSD(amountCents)})`, `${tx.id} on ${targetAcc.accountIdentifier}`);
        showToast('success', 'Withdrawal Completed', `${formatCentsToUSD(amountCents)} debited from ${targetAcc.name}.`);
        return { ok: true, tx };
      };

      const executeTransfer = ({ sourceAccountId, destAccountId, amountCents, description }) => {
        const srcAcc = userAccounts.find(a => a.id === sourceAccountId);
        const dstAcc = userAccounts.find(a => a.id === destAccountId);
        if (!srcAcc || !dstAcc || amountCents > srcAcc.balanceCents) {
          return { ok: false, error: 'Transfer validation failed.' };
        }
        const transferId = generateId('TRF');
        const timeNow = getCurrentChicagoTime12h();
        const newSrcBal = srcAcc.balanceCents - amountCents;
        const newDstBal = dstAcc.balanceCents + amountCents;

        const debitTx = {
          id: generateId('TXN'),
          userId: currentUser.id,
          accountId: srcAcc.id,
          date: CUTOFF_DATE_STR,
          time: timeNow,
          description: `${description} (To ${dstAcc.name})`,
          category: 'Internal transfers',
          direction: 'DEBIT',
          amountCents,
          balanceAfterCents: newSrcBal,
          status: 'Completed'
        };

        const creditTx = {
          id: generateId('TXN'),
          userId: currentUser.id,
          accountId: dstAcc.id,
          date: CUTOFF_DATE_STR,
          time: timeNow,
          description: `${description} (From ${srcAcc.name})`,
          category: 'Internal transfers',
          direction: 'CREDIT',
          amountCents,
          balanceAfterCents: newDstBal,
          status: 'Completed'
        };

        setBankState(prev => ({
          ...prev,
          accounts: prev.accounts.map(a => {
            if (a.id === srcAcc.id) return { ...a, balanceCents: newSrcBal };
            if (a.id === dstAcc.id) return { ...a, balanceCents: newDstBal };
            return a;
          }),
          transactions: [...prev.transactions, debitTx, creditTx]
        }));

        pushNotification(
          currentUser.id,
          'Transfer',
          'Internal Transfer Completed',
          `Transferred ${formatCentsToUSD(amountCents)} from ${srcAcc.name} to ${dstAcc.name}.`
        );
        appendAuditLog(
          currentUser.username,
          'TRANSFER',
          `Atomic Internal Transfer (${formatCentsToUSD(amountCents)})`,
          `${transferId}: ${srcAcc.accountIdentifier} -> ${dstAcc.accountIdentifier}`
        );
        showToast('success', 'Transfer Completed', `${formatCentsToUSD(amountCents)} moved to ${dstAcc.name}.`);
        return { ok: true, transferId, amountCents, debitTx, creditTx };
      };

      /* ==========================================================================
         ROUTE DISPATCHING & ACCESS PROTECTION
         ========================================================================== */
      const isAdminRoute = route.startsWith('/admin');
      if (isAdminRoute) {
        return (
          <>
            <ToastBanner toast={toast} onDismiss={() => setToast(null)} />
            <AdminPortal
              route={route}
              navigate={navigate}
              isAdminAuthenticated={isAdminAuthenticated}
              onAdminLogin={() => {
                setIsAdminAuthenticated(true);
                appendAuditLog('admin@wheatlandbank.com', 'ADMIN_AUTH', 'Administrator Signed In', 'Role: Chief Operations & Compliance Officer');
                navigate('/admin/dashboard');
              }}
              onAdminLogout={() => {
                setIsAdminAuthenticated(false);
                navigate('/');
              }}
              state={bankState}
              onReplyTicket={(ticketId) => {
                setBankState(prev => ({
                  ...prev,
                  supportTickets: prev.supportTickets.map(t =>
                    t.id === ticketId
                      ? {
                          ...t,
                          status: 'Resolved',
                          response: 'Your request has been reviewed and approved by Wheatland Bank Client Operations. All account settings and balances are verified.'
                        }
                      : t
                  )
                }));
                appendAuditLog('admin@wheatlandbank.com', 'SUPPORT', `Resolved Support Ticket ${ticketId}`, 'Sent official response to customer');
                showToast('success', 'Ticket Resolved', `Response sent for ${ticketId}.`);
              }}
            />
          </>
        );
      }

      // Public Routes
      const publicRoutes = ['/', '/register', '/login', '/verify-phone', '/forgot-password', '/reset-password'];
      if (!currentUser) {
        if (!publicRoutes.includes(route)) {
          // Enforce route protection for unauthenticated visitors
          return (
            <>
              <ToastBanner toast={toast} onDismiss={() => setToast(null)} />
              <LoginPage
                navigate={navigate}
                users={bankState.users}
                onCredentialsVerified={(user) => startOtpChallenge(user)}
                onQuickDemoLogin={handleQuickDemoLogin}
                onOpenAdmin={() => navigate('/admin/login')}
              />
            </>
          );
        }

        return (
          <>
            <ToastBanner toast={toast} onDismiss={() => setToast(null)} />
            {route === '/' && (
              <WelcomePage
                navigate={navigate}
                onQuickDemoLogin={handleQuickDemoLogin}
                onOpenAdmin={() => navigate('/admin/login')}
              />
            )}
            {route === '/register' && (
              <RegisterPage
                navigate={navigate}
                existingUsers={bankState.users}
                onRegisterSubmit={handleRegisterSubmit}
              />
            )}
            {route === '/login' && (
              <LoginPage
                navigate={navigate}
                users={bankState.users}
                onCredentialsVerified={(user) => startOtpChallenge(user)}
                onQuickDemoLogin={handleQuickDemoLogin}
                onOpenAdmin={() => navigate('/admin/login')}
              />
            )}
            {route === '/verify-phone' && (
              <PhoneVerificationPage
                navigate={navigate}
                pendingAuth={pendingAuth}
                onVerifySuccess={handleVerifyOtp}
                onResendOtp={() => {
                  if (pendingAuth) {
                    startOtpChallenge(pendingAuth.user);
                    showToast('info', 'New Code Sent', 'A new 6-digit SMS code has been generated.');
                  }
                }}
              />
            )}
            {(route === '/forgot-password' || route === '/reset-password') && (
              <ForgotAndResetPasswordPage
                mode={route === '/reset-password' ? 'reset' : 'forgot'}
                navigate={navigate}
                users={bankState.users}
                onResetPassword={(userId, newPwd) => {
                  setBankState(prev => ({
                    ...prev,
                    users: prev.users.map(u =>
                      u.id === userId ? { ...u, passwordHash: secureHashSim(newPwd) } : u
                    )
                  }));
                  pushNotification(userId, 'Security', 'Password Reset Completed', 'Your online banking password was successfully updated.');
                  appendAuditLog(userId, 'SECURITY', 'Password Reset Executed', 'Verified via security question');
                  showToast('success', 'Password Updated', 'You may now sign in with your new password.');
                  navigate('/login');
                }}
              />
            )}
          </>
        );
      }

      // Authenticated Customer Portal Routes
      const checkingAcc = userAccounts.find(a => a.type === 'Checking') || userAccounts[0];
      const savingsAcc = userAccounts.find(a => a.type === 'Savings') || userAccounts[1];
      const unreadCount = userNotifications.filter(n => !n.read).length;

      return (
        <>
          <ToastBanner toast={toast} onDismiss={() => setToast(null)} />
          <CustomerShell
            route={route}
            navigate={navigate}
            currentUser={currentUser}
            unreadCount={unreadCount}
            onLogout={handleLogout}
          >
            {(route === '/dashboard' || publicRoutes.includes(route)) && (
              <DashboardPage
                navigate={navigate}
                currentUser={currentUser}
                accounts={userAccounts}
                transactions={userTransactions}
              />
            )}

            {route === '/accounts/checking' && checkingAcc && (
              <AccountOverviewPage
                accountType="Checking"
                account={checkingAcc}
                transactions={userTransactions}
                navigate={navigate}
              />
            )}

            {route === '/accounts/savings' && savingsAcc && (
              <AccountOverviewPage
                accountType="Savings"
                account={savingsAcc}
                transactions={userTransactions}
                navigate={navigate}
              />
            )}

            {route === '/account-details' && (
              <AccountDetailsPage
                currentUser={currentUser}
                accounts={userAccounts}
                transactions={userTransactions}
                navigate={navigate}
                onToast={showToast}
              />
            )}

            {route === '/transactions' && (
              <TransactionsPage
                accounts={userAccounts}
                transactions={userTransactions}
              />
            )}

            {route === '/deposits' && (
              <DepositPage
                accounts={userAccounts}
                onExecuteDeposit={executeDeposit}
                navigate={navigate}
              />
            )}

            {route === '/withdrawals' && (
              <WithdrawalPage
                accounts={userAccounts}
                onExecuteWithdrawal={executeWithdrawal}
                navigate={navigate}
              />
            )}

            {route === '/transfers' && (
              <TransferPage
                accounts={userAccounts}
                onExecuteTransfer={executeTransfer}
                navigate={navigate}
              />
            )}

            {route === '/statements' && (
              <StatementsPage
                currentUser={currentUser}
                accounts={userAccounts}
                transactions={userTransactions}
              />
            )}

            {route === '/notifications' && (
              <NotificationsPage
                notifications={userNotifications}
                onMarkRead={(id) =>
                  setBankState(prev => ({
                    ...prev,
                    notifications: prev.notifications.map(n => (n.id === id ? { ...n, read: true } : n))
                  }))
                }
                onMarkAllRead={() =>
                  setBankState(prev => ({
                    ...prev,
                    notifications: prev.notifications.map(n =>
                      n.userId === currentUser.id ? { ...n, read: true } : n
                    )
                  }))
                }
                onDelete={(id) =>
                  setBankState(prev => ({
                    ...prev,
                    notifications: prev.notifications.filter(n => n.id !== id)
                  }))
                }
              />
            )}

            {route === '/profile' && (
              <ProfilePage
                currentUser={currentUser}
                accounts={userAccounts}
                onUpdateProfile={(updates) => {
                  setBankState(prev => ({
                    ...prev,
                    users: prev.users.map(u => (u.id === currentUser.id ? { ...u, ...updates } : u))
                  }));
                  pushNotification(currentUser.id, 'Profile', 'Profile Details Updated', 'Your residential address and contact details have been saved.');
                  appendAuditLog(currentUser.username, 'PROFILE', 'Customer Profile Updated', 'Updated address/contact metadata');
                  showToast('success', 'Profile Saved', 'Your profile updates have been recorded.');
                }}
              />
            )}

            {(route === '/security' || route === '/settings') && (
              <SecurityAndSettingsPage
                mode={route === '/settings' ? 'settings' : 'security'}
                currentUser={currentUser}
                sessions={userSessions}
                onChangePassword={(newPwd) => {
                  setBankState(prev => ({
                    ...prev,
                    users: prev.users.map(u =>
                      u.id === currentUser.id ? { ...u, passwordHash: secureHashSim(newPwd) } : u
                    )
                  }));
                  pushNotification(currentUser.id, 'Security', 'Password Changed', 'Your Wheatland Bank password was updated.');
                  appendAuditLog(currentUser.username, 'SECURITY', 'Password Changed', 'Verified current password before update');
                  showToast('success', 'Password Updated', 'Your new password is active.');
                }}
                onToggleLoginAlerts={() => {
                  const nextVal = !currentUser.loginAlerts;
                  setBankState(prev => ({
                    ...prev,
                    users: prev.users.map(u =>
                      u.id === currentUser.id ? { ...u, loginAlerts: nextVal } : u
                    )
                  }));
                  showToast('info', 'Preference Updated', `Login alerts ${nextVal ? 'enabled' : 'disabled'}.`);
                }}
                onSignOutAllDevices={() => {
                  setBankState(prev => ({
                    ...prev,
                    sessions: prev.sessions.filter(s => s.userId !== currentUser.id || s.current)
                  }));
                  appendAuditLog(currentUser.username, 'SECURITY', 'Signed Out All Other Devices', 'Revoked secondary device tokens');
                  showToast('success', 'Devices Signed Out', 'All other active device sessions have been terminated.');
                }}
              />
            )}

            {route === '/support' && (
              <SupportPage
                tickets={userTickets}
                onCreateTicket={({ subject, message }) => {
                  const newTicket = {
                    id: `SUP-2026-${Math.floor(1000 + Math.random() * 9000)}`,
                    userId: currentUser.id,
                    subject,
                    message,
                    status: 'Open',
                    createdDate: formatDateUS(CUTOFF_DATE_STR),
                    response: null
                  };
                  setBankState(prev => ({
                    ...prev,
                    supportTickets: [newTicket, ...prev.supportTickets]
                  }));
                  appendAuditLog(currentUser.username, 'SUPPORT', `Created Support Ticket ${newTicket.id}`, subject);
                  showToast('success', 'Support Ticket Created', `Reference #${newTicket.id} submitted.`);
                }}
              />
            )}
          </CustomerShell>
        </>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
</body>
